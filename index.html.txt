function restoreRoom() {
  const scene = $('.room-scene');
  if (!scene) return;
  scene.style.background = state.room === 'night' ? 'linear-gradient(#28334f 0 55%,#665d80 55%)' : state.room === 'sunny' ? 'linear-gradient(#f2c77f 0 55%,#d7b68d 55%)' : 'linear-gradient(#9ebbd0 0 55%,#c4b1a0 55%)';

}
